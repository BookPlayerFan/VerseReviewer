Mobile Web App Endpoint with Inlined Data Preloading function doGet() { const template = HtmlService.createTemplateFromFile('Sidebar'); template.initialCards = getDueCards(); return template.evaluate() .setTitle('Verse Review') .addMetaTag('viewport', 'width=device-width, initial-scale=1, maximum-scale=1, user-scalable=no, viewport-fit=cover') .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL); } // 2. Desktop Toolbar Menu function onOpen() { SpreadsheetApp.getUi() .createMenu('Verse Review') .addItem('Open Flashcard Review', 'showFlashcardSidebar') .addToUi(); } // 3. Desktop Sidebar Launcher with Inlined Data Preloading function showFlashcardSidebar() { const template = HtmlService.createTemplateFromFile('Sidebar'); template.initialCards = getDueCards(); const html = template.evaluate() .setTitle('Verse Review') .setWidth(360); SpreadsheetApp.getUi().showSidebar(html); } // Level Configurations const GROUP_CONFIG = { 0: { name: "New", interval: 0, repsNeeded: 1 }, 1: { name: "Daily", interval: 1, repsNeeded: 4 }, 2: { name: "2-Day", interval: 2, repsNeeded: 3 }, 3: { name: "5-Day", interval: 5, repsNeeded: 2 }, 4: { name: "2-Week", interval: 12, repsNeeded: 2 }, 5: { name: "Monthly", interval: 30, repsNeeded: 2 }, 6: { name: "Quarterly", interval: 75, repsNeeded: 1 }, 7: { name: "Half-Year", interval: 180, repsNeeded: 1 }, 8: { name: "Mastery", interval: 360, repsNeeded: 1 } }; /** * Loads all due cards and Level 0 cards from "Verses". */ function getDueCards() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName('Verses'); if (!sheet) return []; const lastRow = sheet.getLastRow(); if (lastRow < 2) return []; const data = sheet.getRange(1, 1, lastRow, 8).getValues(); const today = new Date(); today.setHours(0, 0, 0, 0); const dueCards = []; for (let i = 1; i < data.length; i++) { const nextDueRaw = data[i][1]; // Col B const repRaw = data[i][3]; // Col D const groupRaw = data[i][4]; // Col E const ref = data[i][5]; // Col F const text = data[i][7]; // Col H if (!ref) continue; const groupStr = String(groupRaw).trim(); const isGroupZero = (groupRaw === 0 || groupStr === "0") && groupRaw !== "" && groupRaw !== null; const isOtherGroup = !isGroupZero && groupStr !== "" && !isNaN(Number(groupStr)) && Number(groupStr) >= 1; let isDue = false; let group = 0; let rep = 1; if (isGroupZero) { isDue = true; group = 0; rep = Number(repRaw) || 1; } else if (isOtherGroup) { group = Number(groupStr); rep = Number(repRaw) || 1; if (nextDueRaw instanceof Date) { const d = new Date(nextDueRaw); d.setHours(0, 0, 0, 0); if (d <= today) isDue = true; } else if (typeof nextDueRaw === 'string' && nextDueRaw.trim() !== '') { const parsed = new Date(nextDueRaw); if (!isNaN(parsed.getTime())) { parsed.setHours(0, 0, 0, 0); if (parsed <= today) isDue = true; } } } else { continue; } if (isDue) { const groupInfo = GROUP_CONFIG[group] || GROUP_CONFIG[0]; dueCards.push({ rowIndex: i + 1, reference: ref, text: text || "(No verse text found in Column H)", group: group, groupName: groupInfo.name, rep: rep, repsNeeded: groupInfo.repsNeeded }); } } dueCards.sort((a, b) => a.group - b.group || a.rowIndex - b.rowIndex); return dueCards; } /** * Fast batch write across Columns B:E. */ function recordReview(rowIndex, newNextDateIso, newRep, newGroup) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName('Verses'); if (!sheet) return false; let nextDateVal = ""; let lastDateVal = ""; if (newGroup > 0 && newNextDateIso) { nextDateVal = new Date(newNextDateIso); nextDateVal.setHours(0, 0, 0, 0); lastDateVal = new Date(); lastDateVal.setHours(0, 0, 0, 0); } sheet.getRange(rowIndex, 2, 1, 4).setValues([[nextDateVal, lastDateVal, newRep, newGroup]]); return true; } /** * Adds an unassigned verse to Level 0. */ function addRandomNewVerse() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName('Verses'); if (!sheet) return { success: false, message: 'Sheet "Verses" not found.' }; const lastRow = sheet.getLastRow(); if (lastRow < 2) return { success: false, message: 'No verses found in sheet.' }; const data = sheet.getRange(1, 1, lastRow, 8).getValues(); const candidates = []; for (let i = 1; i < data.length; i++) { const groupRaw = data[i][4]; const ref = data[i][5]; const isBlankGroup = groupRaw === "" || groupRaw === null || groupRaw === undefined; if (ref && isBlankGroup) { candidates.push({ rowIndex: i + 1, reference: ref }); } } if (candidates.length === 0) { return { success: false, message: 'No unassigned verses remaining.' }; } const chosen = candidates[Math.floor(Math.random() * candidates.length)]; sheet.getRange(chosen.rowIndex, 2, 1, 4).setValues([["", "", 1, 0]]); return { success: true, reference: chosen.reference, rowIndex: chosen.rowIndex }; }
